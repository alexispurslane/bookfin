cmake_minimum_required(VERSION 3.22)
project(bookfin_mobi C)
if(NOT DEFINED BOOKFIN_LIBMOBI_SOURCE)
  set(BOOKFIN_LIBMOBI_SOURCE "$ENV{BOOKFIN_LIBMOBI_SOURCE}")
endif()
if(NOT EXISTS "${BOOKFIN_LIBMOBI_SOURCE}/src/mobi.h")
  message(FATAL_ERROR "Use the pinned Bookfin Nix shell for libmobi sources")
endif()
set(USE_ENCRYPTION OFF CACHE BOOL "" FORCE)
set(USE_LIBXML2 OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
set(USE_MINIZ ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${BOOKFIN_LIBMOBI_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/libmobi" EXCLUDE_FROM_ALL)
add_library(bookfin_mobi SHARED convert.c
  "${BOOKFIN_LIBMOBI_SOURCE}/tools/common.c"
  "${BOOKFIN_LIBMOBI_SOURCE}/src/miniz.c")
target_compile_definitions(bookfin_mobi PRIVATE USE_XMLWRITER MINIZ_NO_ZLIB_COMPATIBLE_NAMES=)
target_include_directories(bookfin_mobi PRIVATE
  "${BOOKFIN_LIBMOBI_SOURCE}/src" "${BOOKFIN_LIBMOBI_SOURCE}/tools")
target_link_libraries(bookfin_mobi mobi)
if(ANDROID)
  target_link_options(bookfin_mobi PRIVATE "-Wl,-z,max-page-size=16384")
  target_link_options(mobi PRIVATE "-Wl,-z,max-page-size=16384")
endif()
install(TARGETS bookfin_mobi mobi LIBRARY DESTINATION lib)
